float volume;
int gamme;
int note;
Serial pc(USBTX, USBRX);

PwmOut speacker (p26);
AnalogIn pot1(p19);
AnalogIn pot2(p20);
C12832 lcd(p5,p7,p6,p8,p11);

void LCD_reset(){
    lcd.cls();
    lcd.locate(0,0);
}

void PlayNote(float frequency, float volume){
    speacker.period(1.0/frequency);
    speacker.write(volume);
}

int AnalogIn2Index(float v, int n){
    return floorf(v*n);
}

void detec_chiffre(){
    note = pc.getc();
    PlayNote(freq_notes[note-1],volume);
}

int main(){
    int nbgamme =5;
    gamme=AnalogIn2Index(pot2.read(),nbgamme);
    note =1;
    while (true){
        volume = pot1.read();
        LCD_reset();
        detec_chiffre();
        lcd.printf("Note : %d \n Gamme : %d \n volume : %.2f", note,gamme,volume);
        wait_us(100000);
        speacker.write(0);
    }
}
